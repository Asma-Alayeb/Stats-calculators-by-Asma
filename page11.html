function calculateSampleSize() { const marginErrorPercent = parseFloat(document.getElementById('marginError').value); const confidenceLevel = parseFloat(document.getElementById('ssConfidenceLevel').value) / 100; const stdDev = parseFloat(document.getElementById('ssStdDev').value); // Convert margin of error percent to decimal const marginError = marginErrorPercent / 100; // Validate inputs if ( isNaN(marginError) || isNaN(confidenceLevel) || isNaN(stdDev) || marginError <= 0 || confidenceLevel <= 0 || confidenceLevel >= 1 ) { alert("Please enter valid numbers."); return; } // Full precision z-score const zScore = jStat.normal.inv(1 - (1 - confidenceLevel) / 2, 0, 1); // Sample size formula const n = Math.pow(zScore * stdDev / marginError, 2); // Always round up const sampleSize = Math.ceil(n); // Display result document.getElementById('ssResult').innerHTML = `Required Sample Size: ${sampleSize}`; }
